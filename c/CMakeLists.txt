cmake_minimum_required(VERSION 3.16)
project(tedd_fic_c C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(fic STATIC fic.c)
target_include_directories(fic PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(fic PRIVATE /W4)
else()
  target_compile_options(fic PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(fic_test test.c)
target_link_libraries(fic_test PRIVATE fic)
enable_testing()
add_test(NAME rgb_small COMMAND fic_test roundtrip 1 1 3)
add_test(NAME rgba_strips COMMAND fic_test roundtrip 128 96 4)
add_test(NAME rgb_narrow COMMAND fic_test roundtrip 2 100 3)
add_test(NAME compact_byte_modes COMMAND fic_test fixtures)
